$results = Get-ChildItem "$env:USERPROFILE\.codex\sessions" -Recurse -Filter "rollout-*.jsonl" |
ForEach-Object {
    $file = $_

    $sessionId = [System.IO.Path]::GetFileNameWithoutExtension($file.Name) -replace '^rollout-',''

    $firstUser = Get-Content $file.FullName |
        Select-String '"role":"user"' |
        Select-Object -First 1

    [PSCustomObject]@{
        Date = $file.LastWriteTime
        SessionID = $sessionId
        FirstUserMessage = if ($firstUser) { $firstUser.Line } else { "" }
    }
}

$results |
Sort-Object Date -Descending |
Export-Csv "$env:USERPROFILE\Desktop\codex-sessions.csv" -NoTypeInformation
